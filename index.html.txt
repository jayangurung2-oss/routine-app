<script>
let history = JSON.parse(localStorage.getItem("history")) || [];

function saveDay() {
  let tasks = document.querySelectorAll(".task");
  let done = 0;

  tasks.forEach(t => {
    if (t.checked) done++;
    t.checked = false;
  });

  history.push(done);
  localStorage.setItem("history", JSON.stringify(history));

  drawChart();
}

function drawChart() {
  let good = history.filter(x => x >= 4).length;
  let bad = history.length - good;

  let ctx = document.getElementById("chart").getContext("2d");

  new Chart(ctx, {
    type: "pie",
    data: {
      labels: ["Good Days", "Low Days"],
      datasets: [{
        data: [good, bad],
        backgroundColor: ["#4CAF50", "#FF5252"]
      }]
    }
  });
}

drawChart();
</script>
